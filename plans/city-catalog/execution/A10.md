const report = JSON.parse(await Bun.file("/tmp/geo-backfill.json").text())
const groups = new Map()

for (const item of report.decisions.filter(item => item.status === "needs_review")) {
  const groupKey = JSON.stringify([
    item.sourceLabel,
    item.countryId ?? null,
    item.countryCode ?? null,
  ])

  let group = groups.get(groupKey)
  if (!group) {
    group = {
      city: item.sourceLabel,
      countryId: item.countryId ?? null,
      countryCode: item.countryCode ?? null,
      reason: item.reason,
      candidates: item.candidates.map(candidate => ({
        id: candidate.placeId,
        name: candidate.name,
        country: candidate.countryName,
        region: candidate.regionName,
        matchedName: candidate.matchedName,
      })),
      records: [],
    }
    groups.set(groupKey, group)
  }

  group.records.push({ key: item.key, type: item.entityType })
}

const allGroups = [...groups.values()]
const page = Number(process.env.PAGE ?? 1)
const pageSize = 8

console.log(`Порция ${page}/${Math.ceil(allGroups.length / pageSize)}, всего групп: ${allGroups.length}`)
for (const group of allGroups.slice((page - 1) * pageSize, page * pageSize))
